import java.util.Scanner;
public class home101{

  public static void main (String [] args){
    Scanner sc = new Scanner(System.in);
    //A
    System.out.println("give a number to check prime");
    boolean checkPrime = isPrime(sc.nextInt());
    System.out.println(checkPrime);
    //B
    System.out.println("give a number to check perfect");
    boolean checkPrefect = isPerfect(sc.nextInt());
    System.out.println(checkPrefect);
    //C
    System.out.println("give a number");
    int result = special_sum(sc.nextInt());
    System.out.println(result);

  }
//A
  public static boolean isPrime(int a ){
  boolean flag = true;
  for(int i=2 ; i<a ; i++){
    if( a%i==0){
      flag = false;
    }
  }
  return flag;
  }
  //B
  public static boolean isPerfect(int a ){
    boolean flag = false;
    int sum=0;
    for(int i=1; i<a ;i++){
      if(a%i==0){
        sum +=i;
      }
    }
    if(sum== a ){
      flag = true;
    }
    return flag;
  }
//C
  public static int special_sum (int a ){
    int sum =0;
    for (int i =2 ; i<= a ; i++){
      if( isPrime(i) || isPerfect(i)){
         sum +=i;
      }
   }
    return sum;
  }
}

import java.util.Scanner;

public class home102{

  public static void main (String [] args){
      Scanner sc = new Scanner(System.in);
    //A
      System.out.println("give number to print dots");
    showDot(sc.nextInt());
    System.out.println();
    //B
      System.out.println("give number to print palindrome");
    showPalin(sc.nextInt());
    System.out.println();
    //C
      System.out.println("give number to print diamond");
    showDiamond(sc.nextInt());
  }
  //A
  public static void showDot(int a){
    for (int i =1 ; i<=a ;i++){
      System.out.print(".");
    }

  }
  //B
  public static void showPalin(int a ){
    for (int i =1 ; i<=a ; i++){
      System.out.print(i);
    }
    for (int i = a-1 ; i>0 ; i--){
      System.out.print(i);
    }
  }
//C
  public static void showDiamond (int a ){
      for(int i =1 ; i<=a ; i++){
        showDot(a-i);
        showPalin(i);
        showDot(a-i);
        System.out.println();
      }
      for(int i =a-1 ; i>0 ; i--){
      showDot(a-i);
      showPalin(i);
      showDot(a-i);
      System.out.println();
      }
  }
}
